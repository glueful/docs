# Background Jobs + Notifications

Learn how to process work asynchronously using Glueful's queue system and send notifications via multiple channels. Build an email processing system that handles bulk operations efficiently while keeping your users informed through notifications.

## What You'll Build

In this tutorial, you'll create a robust email processing system that demonstrates:

- **Queue System**: Asynchronous job processing with multiple drivers
- **Background Jobs**: Email processing without blocking the main thread
- **Notification System**: Multi-channel notifications (in-app, email)
- **Error Handling**: Failed job recovery and retry mechanisms
- **Monitoring**: Queue metrics and job status tracking

**Time:** ~35 minutes  
**Level:** Intermediate  
**Topics:** Queue system, Background jobs, Email notifications, Async processing

## Prerequisites

- Glueful installed and configured
- Database connection configured
- Basic understanding of PHP classes and dependency injection
- Email provider configured (SMTP, Sendgrid, or Mailgun)

## Required Extensions

## Step 1: Configure Queue System

### Queue Configuration

Add queue configuration to your `config/queue.php`:

```php
<?php
// config/queue.php

return [
    'default' => env('QUEUE_CONNECTION', 'database'),
    
    'connections' => [
        'database' => [
            'driver' => 'database',
            'table' => 'queue_jobs',
            'failed_table' => 'queue_failed_jobs',
            'retry_after' => 90,
            'max_attempts' => 3
        ],
        
        'redis' => [
            'driver' => 'redis',
            'host' => env('REDIS_HOST', '127.0.0.1'),
            'port' => env('REDIS_PORT', 6379),
            'database' => env('REDIS_DB', 0),
            'prefix' => 'glueful:queue:',
            'retry_after' => 90,
            'job_expiration' => 3600
        ]
    ],
    
    'failed_job_provider' => 'database'
];
```

## Step 2: Create Email Campaign System

Let's create models and jobs for handling email campaigns.

### Campaign Repository

Glueful uses the Repository pattern for data access instead of traditional ORM models:

```php
<?php
// app/Repositories/EmailCampaignRepository.php

namespace App\Repositories;

use Glueful\Database\BaseRepository;
use Glueful\Database\Interfaces\Connection;

class EmailCampaignRepository extends BaseRepository
{
    protected string $table = 'email_campaigns';
    
    // Status constants
    public const STATUS_DRAFT = 'draft';
    public const STATUS_SCHEDULED = 'scheduled';
    public const STATUS_PROCESSING = 'processing';
    public const STATUS_COMPLETED = 'completed';
    public const STATUS_FAILED = 'failed';
    
    public function __construct(Connection $connection)
    {
        parent::__construct($connection);
    }
    
    public function create(array $data): int
    {
        return $this->connection->insert($this->table, [
            'name' => $data['name'],
            'subject' => $data['subject'],
            'template' => $data['template'],
            'template_data' => json_encode($data['template_data'] ?? []),
            'recipient_list' => json_encode($data['recipient_list'] ?? []),
            'status' => $data['status'] ?? self::STATUS_DRAFT,
            'total_recipients' => $data['total_recipients'] ?? 0,
            'sent_count' => 0,
            'failed_count' => 0,
            'scheduled_at' => $data['scheduled_at'] ?? null,
            'user_id' => $data['user_id'],
            'created_at' => date('Y-m-d H:i:s'),
            'updated_at' => date('Y-m-d H:i:s')
        ]);
    }
    
    public function findById(int $id): ?array
    {
        $result = $this->connection->query()
            ->select('*')
            ->from($this->table)
            ->where('id', '=', $id)
            ->get();
            
        if ($result) {
            $campaign = $result[0];
            $campaign['template_data'] = json_decode($campaign['template_data'], true);
            $campaign['recipient_list'] = json_decode($campaign['recipient_list'], true);
            return $campaign;
        }
        
        return null;
    }
    
    public function updateStatus(int $id, string $status): bool
    {
        $updates = ['status' => $status, 'updated_at' => date('Y-m-d H:i:s')];
        
        if ($status === self::STATUS_COMPLETED) {
            $updates['completed_at'] = date('Y-m-d H:i:s');
        }
        
        return $this->connection->update($this->table, $updates, ['id' => $id]) > 0;
    }
    
    public function incrementSentCount(int $id): bool
    {
        return $this->connection->query()
            ->update($this->table)
            ->set('sent_count', 'sent_count + 1')
            ->set('updated_at', date('Y-m-d H:i:s'))
            ->where('id', '=', $id)
            ->execute() > 0;
    }
    
    public function incrementFailedCount(int $id): bool
    {
        return $this->connection->query()
            ->update($this->table)
            ->set('failed_count', 'failed_count + 1')
            ->set('updated_at', date('Y-m-d H:i:s'))
            ->where('id', '=', $id)
            ->execute() > 0;
    }
    
    public function findScheduledCampaigns(): array
    {
        return $this->connection->query()
            ->select('*')
            ->from($this->table)
            ->where('status', '=', self::STATUS_SCHEDULED)
            ->where('scheduled_at', '<=', date('Y-m-d H:i:s'))
            ->get();
    }
}
```

## Step 3: Create Background Jobs

Now let's create the job classes that will handle the actual email processing.

> See also: [Queues & Jobs Guide](/guides/core-services/queues-and-jobs) for retry/backoff patterns and [Notifications Lifecycle](/guides/core-services/notifications#events-integration--lifecycle) for events emitted as notifications move through the system.

### Email Processing Job

```php
<?php
// app/Jobs/ProcessEmailCampaignJob.php

namespace App\Jobs;

use Glueful\Queue\Job;
use Glueful\Notifications\NotificationService;
use App\Repositories\EmailCampaignRepository;
use App\Services\EmailService;
use App\Services\UserService;
use Exception;

class ProcessEmailCampaignJob extends Job
{
    private EmailCampaignRepository $campaignRepository;
    private NotificationService $notificationService;
    private EmailService $emailService;
    private UserService $userService;
    
    public function __construct(array $data = [])
    {
        parent::__construct($data);
        
        // Inject dependencies (in practice, you'd use DI container)
        $this->campaignRepository = app(EmailCampaignRepository::class);
        $this->notificationService = app(NotificationService::class);
        $this->emailService = app(EmailService::class);
        $this->userService = app(UserService::class);
    }
    
    public function handle(): void
    {
        $data = $this->getData();
        $campaignId = $data['campaign_id'];
        $userId = $data['user_id'];
        
        try {
            $campaign = $this->campaignRepository->findById($campaignId);
            if (!$campaign) {
                throw new Exception("Campaign not found: {$campaignId}");
            }
            
            $user = $this->userService->findById($userId);
            
            // Mark campaign as processing
            $this->campaignRepository->updateStatus($campaignId, EmailCampaignRepository::STATUS_PROCESSING);
            
            // Send start notification
            if ($user) {
                $this->sendNotification($user, 'campaign_started', [
                    'campaign_name' => $campaign['name'],
                    'total_recipients' => $campaign['total_recipients']
                ]);
            }
            
            // Process emails in batches
            $this->processEmailBatch($campaign);
            
            // Mark as completed and send completion notification
            $this->campaignRepository->updateStatus($campaignId, EmailCampaignRepository::STATUS_COMPLETED);
            
            if ($user) {
                // Get updated campaign data
                $updatedCampaign = $this->campaignRepository->findById($campaignId);
                $this->sendNotification($user, 'campaign_completed', [
                    'campaign_name' => $updatedCampaign['name'],
                    'sent_count' => $updatedCampaign['sent_count'],
                    'failed_count' => $updatedCampaign['failed_count']
                ]);
            }
            
        } catch (Exception $e) {
            $this->handleJobFailure($campaignId, $userId, $e);
            throw $e; // Re-throw to trigger queue failure handling
        }
    }
    
    private function getData(): array
    {
        return $this->payload['data'] ?? [];
    }
    
    private function processEmailBatch(array $campaign): void
    {
        $batchSize = 50; // Process 50 emails per batch
        $recipients = $campaign['recipient_list'];
        $batches = array_chunk($recipients, $batchSize);
        
        foreach ($batches as $batch) {
            // Queue individual email jobs
            foreach ($batch as $recipient) {
                app(\Glueful\Queue\QueueManager::class)->push(SendEmailJob::class, [
                    'campaign_id' => $campaign['id'],
                    'recipient' => $recipient,
                    'subject' => $campaign['subject'],
                    'template' => $campaign['template'],
                    'template_data' => $campaign['template_data']
                ], 'emails'); // Use dedicated email queue
            }
        }
    }
    
    private function sendNotification(array $user, string $type, array $data): void
    {
        $messages = [
            'campaign_started' => "Email campaign '{$data['campaign_name']}' started processing {$data['total_recipients']} recipients",
            'campaign_completed' => "Email campaign '{$data['campaign_name']}' completed! Sent: {$data['sent_count']}, Failed: {$data['failed_count']}"
        ];
        
        $this->notificationService->send(
            $type,
            $user,
            $messages[$type] ?? 'Campaign notification',
            $data,
            ['channels' => ['database', 'email']]
        );
    }
    
    private function handleJobFailure(int $campaignId, string $userId, Exception $e): void
    {
        try {
            // Update campaign status to failed
            $this->campaignRepository->updateStatus($campaignId, EmailCampaignRepository::STATUS_FAILED);
            
            // Try to send failure notification to user
            $user = $this->userService->findById($userId);
            if ($user) {
                $campaign = $this->campaignRepository->findById($campaignId);
                $this->sendNotification($user, 'campaign_failed', [
                    'campaign_name' => $campaign['name'] ?? 'Unknown',
                    'error' => $e->getMessage()
                ]);
            }
            
        } catch (Exception $notificationError) {
            // Log the notification error but don't throw
            error_log("Failed to send failure notification: " . $notificationError->getMessage());
        }
    }
    
    public function failed(\Exception $exception): void
    {
        // This method is called when the job ultimately fails after retries
        $data = $this->getData();
        
        error_log("Email campaign job failed after all retries: " . $exception->getMessage() . " Data: " . json_encode([
            'campaign_id' => $data['campaign_id'] ?? null,
            'user_id' => $data['user_id'] ?? null,
            'job_uuid' => $this->getUuid()
        ]));
    }
}
```

### Individual Email Job

::callout{type="info"}
Notifications dispatched in this job emit lifecycle events (Queued, Sent, Delivered, Failed). Correlate them with the catalog in [Events & Listeners](/guides/core-services/events-and-listeners#built-in-events--core-listeners) for observability dashboards.
::

```php
<?php
// app/Jobs/SendEmailJob.php

namespace App\Jobs;

use Glueful\Queue\Job;
use App\Repositories\EmailCampaignRepository;
use App\Services\EmailService;
use Exception;

class SendEmailJob extends Job
{
    private EmailCampaignRepository $campaignRepository;
    private EmailService $emailService;
    
    public function __construct(array $data = [])
    {
        parent::__construct($data);
        
        $this->campaignRepository = app(EmailCampaignRepository::class);
        $this->emailService = app(EmailService::class);
    }
    
    public function handle(): void
    {
        $data = $this->getData();
        $campaignId = $data['campaign_id'];
        $recipient = $data['recipient'];
        $subject = $data['subject'];
        $template = $data['template'];
        $templateData = $data['template_data'];
        
        try {
            // Send the email
            $success = $this->emailService->sendTemplateEmail(
                $recipient['email'],
                $subject,
                $template,
                array_merge($templateData, $recipient)
            );
            
            if ($success) {
                // Increment sent count atomically
                $this->campaignRepository->incrementSentCount($campaignId);
            } else {
                throw new Exception("Email service returned false for {$recipient['email']}");
            }
            
        } catch (Exception $e) {
            // Increment failed count
            $this->campaignRepository->incrementFailedCount($campaignId);
            
            error_log("Failed to send email to {$recipient['email']}: " . $e->getMessage());
            throw $e; // Re-throw to trigger retry logic
        }
    }
    
    private function getData(): array
    {
        return $this->payload['data'] ?? [];
    }
    
    public function failed(\Exception $exception): void
    {
        $data = $this->getData();
        $recipient = $data['recipient'] ?? [];
        
        error_log("Email sending failed permanently for {$recipient['email']}: " . $exception->getMessage());
            'recipient' => $recipient['email'] ?? 'unknown',
            'campaign_id' => $data['campaign_id'] ?? null,
            'error' => $exception->getMessage()
        ]);
    }
}
```

## Step 4: Create Email Service

> Tip: High-volume fan-out can hit provider limits—review [Rate Limiting](/guides/core-services/notifications#rate-limiting) before increasing batch sizes.

For production use, you should leverage the `glueful/email-notification` extension instead of creating a custom email service:

::callout{type="info"}
**Recommended Approach:** Use the `glueful/email-notification` extension's `NotificationService` which provides robust multi-provider support, queue integration, and advanced templating. The custom service below is for educational purposes only.
::

> Reference: [Notifications Core Concepts](/guides/core-services/notifications#core-concepts) cover channel abstraction, formatter pipelines, provider failover, and lifecycle events.

```php
<?php
// app/Services/EmailService.php

namespace App\Services;

use Glueful\Notifications\NotificationService;
use Exception;

class EmailService
{
    private NotificationService $notificationService;
    
    public function __construct()
    {
        // Use the email-notification extension's service for production
        $this->notificationService = app(NotificationService::class);
    }
    
    public function sendTemplateEmail(string $to, string $subject, string $template, array $data): bool
    {
        try {
            // Use the notification service from email-notification extension
            return $this->notificationService->sendEmail(
                $to,
                $subject,
                $template,
                $data
            );
            
        } catch (Exception $e) {
            error_log("Email sending failed: " . $e->getMessage());
            throw $e;
        }
    }
    
    // Legacy method for demonstration (not recommended for production)
    public function sendTemplateEmailLegacy(string $to, string $subject, string $template, array $data): bool
    {
        try {
            // Render template
            $body = $this->renderTemplate($template, $data);
            
            // Send via configured provider
            return $this->sendEmail($to, $subject, $body);
            
        } catch (Exception $e) {
            error_log("Email sending failed: " . $e->getMessage());
            throw $e;
        }
    }
    
    private function renderTemplate(string $template, array $data): string
    {
        // Simple template rendering - you might use Twig or similar
        $templatePath = __DIR__ . "/../../resources/email-templates/{$template}.php";
        
        if (!file_exists($templatePath)) {
            throw new Exception("Template not found: {$template}");
        }
        
        // Extract variables for template
        extract($data);
        
        ob_start();
        include $templatePath;
        return ob_get_clean();
    }
    
    private function sendEmail(string $to, string $subject, string $body): bool
    {
        $provider = $this->config['default'] ?? 'smtp';
        
        return match ($provider) {
            'smtp' => $this->sendViaSmtp($to, $subject, $body),
            'sendgrid' => $this->sendViaSendgrid($to, $subject, $body),
            'mailgun' => $this->sendViaMailgun($to, $subject, $body),
            default => throw new Exception("Unsupported email provider: {$provider}")
        };
    }
    
    private function sendViaSmtp(string $to, string $subject, string $body): bool
    {
        $config = $this->config['providers']['smtp'];
        
        // Use PHP's mail() function or a library like SwiftMailer/Symfony Mailer
        $headers = [
            'From: ' . $config['from_address'],
            'Reply-To: ' . $config['from_address'],
            'Content-Type: text/html; charset=UTF-8'
        ];
        
        return mail($to, $subject, $body, implode("\r\n", $headers));
    }
    
    private function sendViaSendgrid(string $to, string $subject, string $body): bool
    {
        $config = $this->config['providers']['sendgrid'];
        
        $data = [
            'personalizations' => [
                ['to' => [['email' => $to]]]
            ],
            'from' => ['email' => $config['from']['address'], 'name' => $config['from']['name']],
            'subject' => $subject,
            'content' => [
                ['type' => 'text/html', 'value' => $body]
            ]
        ];
        
        $ch = curl_init();
        curl_setopt_array($ch, [
            CURLOPT_URL => 'https://api.sendgrid.com/v3/mail/send',
            CURLOPT_POST => true,
            CURLOPT_POSTFIELDS => json_encode($data),
            CURLOPT_HTTPHEADER => [
                'Authorization: Bearer ' . $config['api_key'],
                'Content-Type: application/json'
            ],
            CURLOPT_RETURNTRANSFER => true
        ]);
        
        $response = curl_exec($ch);
        $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        curl_close($ch);
        
        return $httpCode >= 200 && $httpCode < 300;
    }
    
    private function sendViaMailgun(string $to, string $subject, string $body): bool
    {
        $config = $this->config['providers']['mailgun'];
        
        $data = [
            'from' => $config['from'],
            'to' => $to,
            'subject' => $subject,
            'html' => $body
        ];
        
        $ch = curl_init();
        curl_setopt_array($ch, [
            CURLOPT_URL => $config['endpoint'] . '/messages',
            CURLOPT_POST => true,
            CURLOPT_POSTFIELDS => http_build_query($data),
            CURLOPT_USERPWD => 'api:' . $config['api_key'],
            CURLOPT_RETURNTRANSFER => true
        ]);
        
        $response = curl_exec($ch);
        $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        curl_close($ch);
        
        return $httpCode >= 200 && $httpCode < 300;
    }
}
```

```php
<?php
// app/Controllers/EmailCampaignController.php

namespace App\Controllers;

use App\Repositories\EmailCampaignRepository;

use Glueful\Queue\QueueManager;

class EmailCampaignController
{
    public function __construct(
        private EmailCampaignRepository $campaignRepository,
        private QueueManager $queueManager,
        private Validator $validator
    ) {}
    
    public function index(Request $request): Response
    {
        // For simplicity, we'll get all campaigns - you'd implement pagination in practice
        $campaigns = $this->campaignRepository->findAll();
            
        return Response::json([
            'campaigns' => $campaigns
        ]);
    }
    
    public function store(Request $request): Response
    {
        $rules = [
            'name' => 'required|string|max:255',
            'subject' => 'required|string|max:255',
            'template' => 'required|string|max:100',
            'template_data' => 'array',
            'recipient_list' => 'required|array|min:1',
            'recipient_list.*.email' => 'required|email',
            'recipient_list.*.name' => 'string|max:255',
            'scheduled_at' => 'date|after:now'
        ];
        
        $data = $this->validator->validate($request->request->all(), $rules);
        
        // Calculate total recipients
        $data['total_recipients'] = count($data['recipient_list']);
        $data['user_id'] = auth()->user()->id; // Assuming auth middleware
        
        $campaignId = $this->campaignRepository->create($data);
        $campaign = $this->campaignRepository->findById($campaignId);
        
        return Response::json([
            'message' => 'Email campaign created successfully',
            'campaign' => $campaign
        ], 201);
    }
    
    public function show(Request $request): Response
    {
        $id = $request->attributes->get('id');
        $campaign = $this->campaignRepository->findById($id);
        
        if (!$campaign) {
            return Response::json(['error' => 'Campaign not found'], 404);
        }
        
        return Response::json(['campaign' => $campaign]);
    }
    
    public function send(Request $request): Response
    {
        $id = $request->attributes->get('id');
        $campaign = $this->campaignRepository->findById($id);
        
        if (!$campaign) {
            return Response::json(['error' => 'Campaign not found'], 404);
        }
        
        if ($campaign['status'] === EmailCampaignRepository::STATUS_PROCESSING) {
            return Response::json([
                'error' => 'Campaign is already being processed'
            ], 422);
        }
        
        // Queue the campaign processing job
        $jobId = $this->queueManager->push(ProcessEmailCampaignJob::class, [
            'campaign_id' => $campaign['id'],
            'user_id' => auth()->user()->id
        ], 'campaigns');
        
        $this->campaignRepository->updateStatus($campaign['id'], EmailCampaignRepository::STATUS_SCHEDULED);
        
        return Response::json([
            'message' => 'Email campaign queued for processing',
            'job_id' => $jobId,
            'campaign' => $this->campaignRepository->findById($campaign['id'])
        ]);
    }
    
    public function stats(Request $request): Response
    {
        $id = $request->attributes->get('id');
        $campaign = $this->campaignRepository->findById($id);
        
        if (!$campaign) {
            return Response::json(['error' => 'Campaign not found'], 404);
        }
        
        $stats = [
            'total_recipients' => $campaign['total_recipients'],
            'sent_count' => $campaign['sent_count'],
            'failed_count' => $campaign['failed_count'],
            'pending_count' => $campaign['total_recipients'] - $campaign['sent_count'] - $campaign['failed_count'],
            'success_rate' => $campaign['sent_count'] > 0 
                ? round(($campaign['sent_count'] / $campaign['total_recipients']) * 100, 2)
                : 0,
            'status' => $campaign['status'],
            'created_at' => $campaign['created_at'],
            'completed_at' => $campaign['completed_at']
        ];
        
        return Response::json(['stats' => $stats]);
    }
}
```

### Queue Status Endpoint

```php
<?php
// app/Controllers/QueueStatusController.php

namespace App\Controllers;

use Glueful\Http\Response;
use Glueful\Queue\QueueManager;
use Symfony\Component\HttpFoundation\Request;

class QueueStatusController
{
    public function __construct(private QueueManager $queueManager) {}
    
    public function status(): Response
    {
        $stats = [
            'queues' => [],
            'failed_jobs' => 0,
            'total_jobs' => 0
        ];
        
        // Get stats for each queue
        $queues = ['default', 'campaigns', 'emails'];
        foreach ($queues as $queueName) {
            $queueStats = $this->queueManager->getQueueStats($queueName);
            $stats['queues'][$queueName] = $queueStats;
            $stats['total_jobs'] += $queueStats['pending'] ?? 0;
        }
        
        // Get failed jobs count
        $stats['failed_jobs'] = $this->queueManager->getFailedJobsCount();
        
        return Response::json($stats);
    }
    
    public function retryFailed(): Response
    {
        $retriedCount = $this->queueManager->retryFailedJobs();
        
        return Response::json([
            'message' => "Retried {$retriedCount} failed jobs"
        ]);
    }
}

        $this->info('Queue Status Report');
