Glueful provides comprehensive database performance monitoring tools to help you track, analyze, and optimize your database operations in real-time.

## Overview

The performance monitoring system includes:

- **Real-time Metrics**: Live database performance statistics
- **Query Performance Tracking**: Detailed query execution monitoring
- **Resource Usage Monitoring**: Memory, CPU, and I/O tracking
- **Health Checks**: Automated database health assessment
- **Alerting System**: Configurable performance alerts

## Real-Time Performance Monitoring

### Basic Monitoring Setup

```php
use Glueful\Database\PerformanceMonitor;
use Glueful\Database\Connection;

$connection = new Connection();
$monitor = new PerformanceMonitor($connection);

// Start monitoring
$monitor->startMonitoring();

// Execute your database operations
$users = $db->table('users')->where('status', 'active')->get();
$orders = $db->table('orders')->where('created_at', '>', '2025-01-01')->get();

// Get performance metrics
$metrics = $monitor->getMetrics();

echo "Queries Executed: {$metrics['query_count']}\n";
echo "Total Execution Time: {$metrics['total_time']}ms\n";
echo "Average Query Time: {$metrics['avg_time']}ms\n";
echo "Memory Usage: " . formatBytes($metrics['memory_usage']) . "\n";
echo "Cache Hit Ratio: {$metrics['cache_hit_ratio']}%\n";
```

### Advanced Monitoring Configuration

```php
// Configure monitoring with custom thresholds
$monitor = new PerformanceMonitor($connection, [
    'slow_query_threshold' => 1000,     // milliseconds
    'memory_threshold' => 128 * 1024 * 1024, // 128MB
    'connection_threshold' => 80,        // 80% of max connections
    'sampling_rate' => 0.1,             // Monitor 10% of queries
    'enable_stack_traces' => true,      // Capture stack traces for slow queries
]);

// Set up performance alerts
$monitor->onSlowQuery(function($query, $time, $stackTrace) {
    logger()->warning("Slow query detected", [
        'query' => $query,
        'execution_time' => $time,
        'stack_trace' => $stackTrace
    ]);
});

$monitor->onHighMemoryUsage(function($usage, $threshold) {
    logger()->alert("High memory usage detected", [
        'current_usage' => $usage,
        'threshold' => $threshold
    ]);
});
```

## Query Performance Tracking

### Detailed Query Analysis

```php
use Glueful\Database\QueryPerformanceTracker;

$tracker = new QueryPerformanceTracker();

// Track specific operations
$tracker->startOperation('user_dashboard');

// Your database operations
$user = $userRepo->find($userId);
$orders = $orderRepo->getRecentOrders($userId);
$analytics = $analyticsRepo->getUserStats($userId);

$tracker->endOperation('user_dashboard');

// Get operation performance data
$operationData = $tracker->getOperationData('user_dashboard');

echo "Operation: user_dashboard\n";
echo "Total Time: {$operationData['total_time']}ms\n";
echo "Query Count: {$operationData['query_count']}\n";
echo "Memory Peak: " . formatBytes($operationData['memory_peak']) . "\n";

// Get individual query performance
foreach ($operationData['queries'] as $query) {
    echo "\nQuery: {$query['sql']}\n";
    echo "Time: {$query['time']}ms\n";
    echo "Rows: {$query['rows_affected']}\n";
    echo "Memory: " . formatBytes($query['memory_used']) . "\n";
}
```

### Query Pattern Analysis

```php
// Analyze query patterns over time
$patterns = $tracker->analyzeQueryPatterns([
    'time_window' => '1 hour',
    'group_by' => 'operation',
    'include_stack_traces' => false
]);

foreach ($patterns as $pattern) {
    echo "Pattern: {$pattern['operation']}\n";
    echo "Frequency: {$pattern['count']} times\n";
    echo "Avg Time: {$pattern['avg_time']}ms\n";
    echo "Total Time: {$pattern['total_time']}ms\n";
    echo "Slowest: {$pattern['max_time']}ms\n\n";
}
```

## Resource Usage Monitoring

### Memory Usage Tracking

```php
use Glueful\Database\ResourceMonitor;

$resourceMonitor = new ResourceMonitor();

// Monitor memory usage during database operations
$resourceMonitor->startMemoryTracking();

// Execute memory-intensive operations
$largeDataset = $db->table('large_table')->get();
$processedData = processLargeDataset($largeDataset);

$memoryUsage = $resourceMonitor->getMemoryUsage();

echo "Memory Usage:\n";
echo "Peak Usage: " . formatBytes($memoryUsage['peak']) . "\n";
echo "Current Usage: " . formatBytes($memoryUsage['current']) . "\n";
echo "Memory Limit: " . formatBytes($memoryUsage['limit']) . "\n";
echo "Usage Percentage: {$memoryUsage['percentage']}%\n";

// Check for memory warnings
if ($memoryUsage['percentage'] > 80) {
    echo "⚠️  High memory usage detected!\n";
    
    // Get memory allocation breakdown
    $breakdown = $resourceMonitor->getMemoryBreakdown();
    foreach ($breakdown as $component => $usage) {
        echo "{$component}: " . formatBytes($usage) . "\n";
    }
}
```

### Connection Pool Monitoring

```php
use Glueful\Database\ConnectionPoolMonitor;

$poolMonitor = new ConnectionPoolMonitor();

// Get real-time pool statistics

echo "Connection Pool Status:\n";
```

## Health Checks and Monitoring

### Database Health Assessment

```php
use Glueful\Database\HealthChecker;

$healthChecker = new HealthChecker($connection);

// Perform comprehensive health check
$healthReport = $healthChecker->performHealthCheck();

echo "Database Health Report:\n";
echo "Overall Status: {$healthReport['status']}\n";
echo "Score: {$healthReport['score']}/100\n\n";

foreach ($healthReport['checks'] as $check) {
    $status = $check['passed'] ? '✅' : '❌';
    echo "{$status} {$check['name']}: {$check['message']}\n";
    
    if (!$check['passed'] && isset($check['recommendation'])) {
        echo "   Recommendation: {$check['recommendation']}\n";
    }
}
```

### Automated Health Monitoring

```php
// Set up automated health monitoring
$healthChecker->scheduleHealthChecks([
    'interval' => '5 minutes',
    'notify_on_failure' => true,
    'email_alerts' => ['admin@example.com'],
    'slack_webhook' => 'https://hooks.slack.com/...',
]);

// Configure health check rules
$healthChecker->addCustomCheck('connection_response_time', function($connection) {
    $start = microtime(true);
    $connection->query("SELECT 1");
    $responseTime = (microtime(true) - $start) * 1000;
    
    return [
        'passed' => $responseTime < 100,
        'message' => "Database response time: {$responseTime}ms",
        'value' => $responseTime
    ];
});
```

## CLI Performance Monitoring Commands

Glueful provides CLI commands for database performance monitoring:

```bash
# Real-time performance monitoring
php glueful db:monitor --realtime

# Generate performance report
php glueful db:performance:report --period="last 24 hours" --format=json

# Monitor specific metrics
php glueful db:monitor --metric=query_time --threshold=1000

# Check database health
php glueful db:health:check --detailed

# Monitor connection pool
php glueful db:pool:monitor --interval=5

# Profile application performance
php glueful performance:profile --database-only
```

### Performance Reporting

```bash
# Generate detailed performance report
php glueful db:performance:report \
    --start="2025-01-01 00:00:00" \
    --end="2025-01-31 23:59:59" \
    --output=performance_report.pdf \
    --include-graphs

# Export metrics for external monitoring
php glueful db:metrics:export \
    --format=prometheus \
    --output=metrics.txt

# Compare performance across time periods
php glueful db:performance:compare \
    --period1="last week" \
    --period2="this week" \
    --metrics=query_time,memory_usage,cache_hit_ratio
```

## Performance Metrics and KPIs

### Key Performance Indicators

```php
use Glueful\Database\MetricsCollector;

$metrics = new MetricsCollector();

// Collect key performance indicators
$kpis = $metrics->getKPIs([
    'time_window' => '1 hour',
    'include_trends' => true
]);

echo "Database KPIs (Last Hour):\n";
echo "Average Query Time: {$kpis['avg_query_time']}ms\n";
echo "Queries per Second: {$kpis['queries_per_second']}\n";
echo "Cache Hit Ratio: {$kpis['cache_hit_ratio']}%\n";
echo "Error Rate: {$kpis['error_rate']}%\n";
echo "Connection Utilization: {$kpis['connection_utilization']}%\n";
echo "Throughput: {$kpis['throughput']} operations/sec\n";

// Check performance trends
if ($kpis['trends']['avg_query_time'] > 0.1) {
    echo "⚠️  Query performance degrading (+{$kpis['trends']['avg_query_time']}%)\n";
}

if ($kpis['trends']['cache_hit_ratio'] < -0.05) {
    echo "⚠️  Cache performance declining ({$kpis['trends']['cache_hit_ratio']}%)\n";
}
```

### Custom Metrics

```php
// Define custom metrics
$metrics->defineCustomMetric('user_login_queries', function() use ($db) {
    $start = microtime(true);
    $result = $db->table('users')
        ->where('last_login', '>', date('Y-m-d H:i:s', strtotime('-1 hour')))
        ->count();
    $duration = microtime(true) - $start;
    
    return [
        'value' => $result,
        'duration' => $duration * 1000,
        'timestamp' => time()
    ];
});

// Collect custom metrics
$customMetrics = $metrics->collectCustomMetrics();
```

## Alerting and Notifications

### Performance Alerts

```php
use Glueful\Database\AlertManager;

$alertManager = new AlertManager();

// Configure performance thresholds
$alertManager->addThreshold('slow_query', [
    'metric' => 'query_execution_time',
    'threshold' => 2000, // 2 seconds
    'severity' => 'warning',
    'notification_channels' => ['email', 'slack']

$alertManager->addThreshold('high_memory', [
    'metric' => 'memory_usage_percentage',
    'threshold' => 85, // 85%
    'severity' => 'critical',
    'notification_channels' => ['email', 'slack', 'pagerduty']
]);

$alertManager->addThreshold('connection_pool_exhaustion', [
    'metric' => 'pool_utilization',
    'threshold' => 90, // 90%
    'severity' => 'critical',
    'notification_channels' => ['email', 'slack', 'pagerduty']
]);

// Set up notification channels
$alertManager->configureEmail([
    'smtp_host' => 'smtp.example.com',
    'recipients' => ['dba@example.com', 'dev-team@example.com']
]);

$alertManager->configureSlack([
    'webhook_url' => 'https://hooks.slack.com/...',
    'channel' => '#database-alerts'
]);
```

### Alert Rules

## Best Practices

### Production Monitoring

```php
// Production monitoring setup
class ProductionDatabaseMonitoring 
{
    public function setupMonitoring()
    {
        $monitor = new PerformanceMonitor($this->connection, [
            'sampling_rate' => 0.05, // 5% sampling in production
            'slow_query_threshold' => 500,
            'memory_threshold' => 256 * 1024 * 1024,
            'enable_stack_traces' => false, // Disable for performance
        ]);
        
        // Critical alerts only
        $monitor->onCriticalIssue(function($issue) {
            $this->alertManager->sendCriticalAlert($issue);
        });
        
        return $monitor;
    }
}
```

### Development Monitoring

```php
// Development monitoring setup
if (config('app.env') === 'development') {
    $monitor = new PerformanceMonitor($connection, [
        'sampling_rate' => 1.0, // Monitor all queries
        'slow_query_threshold' => 100,
        'enable_stack_traces' => true,
        'detailed_logging' => true
    ]);
    
    // Log all performance issues
    $monitor->onPerformanceIssue(function($issue) {
        logger()->info('Performance issue detected', $issue);
    });
}
```
