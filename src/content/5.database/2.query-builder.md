Glueful provides a powerful fluent query builder that simplifies database operations with a clean, chainable API. The QueryBuilder offers database-agnostic operations with robust security features, comprehensive logging, and advanced query capabilities.

## Basic Usage

The query builder allows for expressive, readable database operations:

```php
use Glueful\Database\Connection;
use Glueful\Database\QueryBuilder;

// Initialize query builder
$connection = new Connection();
$db = new QueryBuilder($connection->getPDO(), $connection->getDriver());

// Or get it from the container
$db = container()->get(DatabaseInterface::class);

// Simple select query
$users = $db->select('users', ['id', 'name', 'email'])
    ->where(['status' => 'active'])
    ->orderBy(['name' => 'ASC'])
    ->limit(10)
    ->get();
```

## Select Operations

Retrieve data from database tables with various conditions:

```php
// Basic select with columns
$users = $db->select('users', ['id', 'name', 'email'])->get();

// Select with conditions
$activeUsers = $db->select('users', ['*'], ['status' => 'active'])->get();

// Complex conditions with where methods
$posts = $db->select('posts', ['id', 'title', 'created_at'])
    ->where(['status' => 'published'])
    ->whereGreaterThanOrEqual('created_at', '2025-01-01')
    ->get();

// Selecting with joins
$data = $db->select('users', ['users.id', 'users.name', 'posts.title'])
    ->join('posts', 'posts.user_id = users.id', 'INNER')
    ->where(['users.status' => 'active'])
    ->get();
```

## Insert Operations

```php
// Single row insert
$affectedRows = $db->insert('users', [
    'uuid' => Utils::generateNanoID(),
    'name' => 'John Doe',
    'email' => 'john@example.com',
    'status' => 'active'
]);

// Batch insert
$affectedRows = $db->insertBatch('users', [
    [
        'uuid' => Utils::generateNanoID(),
        'name' => 'Jane Smith',
        'email' => 'jane@example.com',
        'status' => 'active'
    ],
    [
        'uuid' => Utils::generateNanoID(),
        'name' => 'Mike Johnson',
        'email' => 'mike@example.com',
        'status' => 'inactive'
    ]
]);

// Get last inserted ID
$lastId = $db->getConnection()->getPDO()->lastInsertId();
```

## Update Operations

```php
// Basic update
$affected = $db->update('users',
    ['status' => 'inactive'],
    ['id' => 5]
);

// Update with conditions
$affected = $db->update('users',
    ['status' => 'inactive'],
    ['last_login' => ['<', date('Y-m-d', strtotime('-1 year'))]]
);

// Update with raw expressions
$db->update('products', 
    ['stock' => 'stock - 1'], 
    ['id' => $productId]
);
```

## Delete Operations

Glueful supports both hard and soft deletes:

```php
// Hard delete with conditions
$affected = $db->delete('users', ['status' => 'inactive'], false);

// Soft delete (sets deleted_at timestamp)
$affected = $db->delete('posts', ['id' => $postId], true);

// Delete with complex conditions
$affected = $db->delete('posts', [
    'created_at' => ['<', date('Y-m-d', strtotime('-1 year'))]
], false);

// Restore soft-deleted records (if restore method exists)
$affected = $db->update('users', 
    ['deleted_at' => null], 
    ['id' => $userId, 'deleted_at' => ['IS NOT', null]]
);
```

## Advanced Queries

### Where Clauses

```php
// Multiple conditions (AND)
$users = $db->select('users', ['*'])
    ->where(['status' => 'active', 'role' => 'admin'])
    ->get();

// OR conditions using whereOr
$users = $db->select('users', ['*'])
    ->where(['status' => 'active'])
    ->whereOr(function($q) {
        $q->where(['role' => 'admin']);
    })
    ->get();

// Where In
$users = $db->select('users', ['*'])
    ->whereIn('id', [1, 2, 3, 4])
    ->get();

// Where not in
$products = $db->select('products', ['*'])
    ->whereNotIn('category_id', [5, 6, 7])
    ->get();

// Where null/not null
$activeUsers = $db->select('users', ['*'])
    ->whereNull('deleted_at')
    ->whereNotNull('last_login')
    ->get();

// Where between
$recentPosts = $db->select('posts', ['*'])
    ->whereBetween('created_at', '2025-01-01', '2025-12-31')
    ->get();

// Where like
$users = $db->select('users', ['*'])
    ->whereLike('name', '%Smith%')
    ->get();

// Greater than / Less than
$youngUsers = $db->select('users', ['*'])
    ->whereGreaterThan('age', 18)
    ->whereLessThan('age', 65)
    ->get();

// Raw WHERE conditions
$yearUsers = $db->select('users', ['*'])
    ->whereRaw('YEAR(created_at) = ?', [2025])
    ->get();
```

### Joins

```php
// Inner join
$usersWithProfiles = $db->select('users', ['users.*', 'profiles.biography'])
    ->join('profiles', 'users.id = profiles.user_id', 'INNER')
    ->get();

// Left join
$usersWithPosts = $db->select('users', ['users.name', 'posts.title'])
    ->join('posts', 'users.id = posts.user_id', 'LEFT')
    ->get();

// Multiple joins
$orderDetails = $db->select('orders', ['orders.id', 'users.name', 'products.title'])
    ->join('users', 'orders.user_id = users.id', 'INNER')
    ->join('products', 'orders.product_id = products.id', 'LEFT')
    ->get();

// Right join (if supported)
$allPostsWithUsers = $db->select('users', ['users.name', 'posts.title'])
    ->join('posts', 'users.id = posts.user_id', 'RIGHT')
    ->get();
```

### Aggregates

```php
// Count
$count = $db->count('users', ['status' => 'active']);

// Count with conditions
$activeUserCount = $db->select('users', ['COUNT(*) as count'])
    ->where(['status' => 'active'])
    ->first();

// Max, Min, Average using raw queries or specific methods if available
$maxAge = $db->select('users', ['MAX(age) as max_age'])->first();
$minAge = $db->select('users', ['MIN(age) as min_age'])->first();
$avgAge = $db->select('users', ['AVG(age) as avg_age'])->first();

// Sum
$totalSales = $db->select('orders', ['SUM(amount) as total'])
    ->where(['status' => 'completed'])
    ->first();

// Multiple aggregates
$userStats = $db->select('users', [
    'COUNT(*) as total_users',
    'AVG(age) as average_age',
    'MAX(created_at) as latest_signup'
])->first();
```

### Grouping and Having

```php
// Group by with having
$usersByStatus = $db->select('users', ['status', 'COUNT(*) as user_count'])
    ->groupBy(['status'])
    ->having(['user_count' => ['>', 10]])
    ->get();

// Group by with having raw conditions
$productsByCategory = $db->select('products', ['category_id', 'AVG(price) as avg_price'])
    ->groupBy(['category_id'])
    ->whereRaw('AVG(price) > ?', [100])
    ->get();

// Multiple group by columns
$salesStats = $db->select('orders', [
    'YEAR(created_at) as year',
    'MONTH(created_at) as month',
    'COUNT(*) as order_count',
    'SUM(total) as revenue'
])
    ->groupBy(['YEAR(created_at)', 'MONTH(created_at)'])
    ->orderBy(['year' => 'ASC', 'month' => 'ASC'])
    ->get();
```

## Raw Expressions

```php
);
```

## Executing Raw SQL Queries

Sometimes you need to execute custom SQL queries that can't be easily expressed using the query builder. Glueful provides the `rawQuery()` method for these scenarios:

```php
// Execute a custom SQL query with parameter binding
$results = $db->rawQuery(
    "SELECT p.*, c.name as category_name 
     FROM products p 
     JOIN categories c ON p.category_id = c.id 
     WHERE p.price > ? AND c.active = ?",
    [100, true]
);

// Execute a complex query with joins and subqueries
$analytics = $db->rawQuery("
    SELECT 
        DATE_FORMAT(o.created_at, '%Y-%m') as month,
        SUM(o.total) as revenue,
        COUNT(DISTINCT o.customer_id) as unique_customers
    FROM orders o
    WHERE o.status = 'completed'
    AND o.created_at >= DATE_SUB(NOW(), INTERVAL 6 MONTH)
    GROUP BY month
    ORDER BY month ASC
");

// Execute a database-specific query
$db->rawQuery("
    SET @rank = 0;
    SELECT 
        @rank := @rank + 1 as rank,
        p.name,
        p.views
    FROM products p
    ORDER BY p.views DESC
    LIMIT 10
");
```

The `rawQuery()` method offers several advantages:

1. **Parameter Binding**: Automatically binds parameters to protect against SQL injection
2. **Error Handling**: Includes comprehensive error handling and logging
3. **Cross-Database Support**: Works across supported database platforms
4. **Performance Tracking**: Integrates with the QueryLogger for query performance analysis

### When to Use Raw Queries

While the fluent query builder covers most use cases, raw queries are appropriate for:

- Complex analytical queries with advanced SQL features
- Database-specific optimizations and hints
- Dynamic SQL generation where the structure varies significantly
- Queries that use special functions not supported by the query builder

## Pagination

Glueful provides built-in pagination support:

```php
// Get paginated results
$page = $_GET['page'] ?? 1;
$perPage = $_GET['per_page'] ?? 20;

$result = $db->select('products', ['*'])
    ->where(['category_id' => $categoryId])
    ->orderBy(['created_at' => 'DESC'])
    ->paginate($page, $perPage);

// $result contains:
// - data: Array of records for the current page
// - current_page: Current page number
// - per_page: Records per page
// - total: Total record count
// - last_page: Number of pages
// - has_more: Whether there are more pages
// - from: First record index
// - to: Last record index

// Alternative pagination syntax
$paginatedUsers = $db->select('users', ['id', 'username', 'email'])
    ->where(['status' => 'active'])
    ->orderBy(['username' => 'ASC'])
    ->limit($perPage)
    ->offset(($page - 1) * $perPage)
    ->get();
```

## Transactions

Glueful provides robust transaction support:

```php
// Simple transaction using PDO
$connection = $db->getConnection();
$pdo = $connection->getPDO();

try {
    $pdo->beginTransaction();

    // Perform multiple operations
    $db->insert('users', [
        'uuid' => Utils::generateNanoID(),
        'name' => 'John Doe',
        'email' => 'john@example.com',
    ]);

    $db->insert('profiles', [
        'user_uuid' => $userUuid,
        'bio' => 'New user bio',
    ]);

    $pdo->commit();
} catch (\Exception $e) {
    $pdo->rollback();
    throw $e;
}

// Transaction with callback (if available)
$result = $db->transaction(function() use ($db) {
    // Perform multiple operations within transaction
    $orderId = $db->insert('orders', [
        'customer_id' => 123,
        'total' => 99.99
    ]);
    
    $db->insert('order_items', [
        'order_id' => $orderId,
        'product_id' => 456,
        'quantity' => 1,
        'price' => 99.99
    ]);
    
    // Return result from transaction
    return $orderId;
});
```

## Single Record Operations

## Debug and Logging

Glueful's QueryBuilder includes built-in query logging capabilities that can be useful during development:

```php
// Enable debug mode (if available)
if (method_exists($db, 'enableDebug')) {
    $db->enableDebug(true);

// Access query logging through separate logger
use Glueful\Database\QueryLogger;

$queryLogger = new QueryLogger();

// Get query statistics
$stats = $queryLogger->getStatistics();
```

## Best Practices

1. **Use Parameter Binding**: Always use parameter binding instead of string concatenation to protect against SQL injection.

```php
// GOOD (parameter binding)
$db->where(['email' => $email]);
$db->rawQuery("SELECT * FROM users WHERE email = ?", [$email]);

// BAD (vulnerable to SQL injection)
$db->whereRaw("email = '" . $email . "'");
$db->rawQuery("SELECT * FROM users WHERE email = '" . $email . "'");
```

2. **Use Transactions**: Wrap multiple related operations in transactions to ensure database consistency.

3. **Consider Soft Deletes**: Use soft deletes for important data to prevent permanent data loss.

4. **Optimize Pagination**: Be mindful of pagination with large datasets. Consider using indexed columns.

5. **DB-Agnostic Queries**: Use the query builder's database-agnostic features instead of writing raw SQL when possible.

6. **Prefer Query Builder Over Raw SQL**: When possible, use the query builder instead of raw SQL for better maintainability and security.

## Advanced QueryBuilder Features

Glueful's QueryBuilder includes several advanced features that enhance performance and provide additional functionality beyond basic CRUD operations.

### Query Caching

The QueryBuilder supports automatic result caching to improve performance for frequently executed queries:

```php
// Cache query results for 1 hour (3600 seconds)
$users = $db->select('users', ['id', 'name', 'email'])
    ->where(['status' => 'active'])
    ->cache(3600)
    ->get();

// Cache with custom cache key
$products = $db->select('products', ['*'])
    ->where(['category_id' => $categoryId])
    ->cache(1800, "products_category_{$categoryId}")
    ->get();

// Disable caching for specific query (when global caching is enabled)
$freshData = $db->select('users', ['*'])
    ->noCache()
    ->get();
```

### Multi-Column Search

Perform searches across multiple columns with various operators:

```php
// Search across multiple columns with OR logic
$results = $db->select('users', ['*'])
    ->search(['name', 'email', 'username'], 'john', 'OR')
    ->get();

// Search with AND logic (all columns must match)
$results = $db->select('products', ['*'])
    ->search(['name', 'description'], 'smartphone', 'AND')
    ->get();

// Manual search with multiple LIKE conditions
$results = $db->select('users', ['*'])
    ->whereLike('name', '%john%')
    ->whereOr(function($q) {
        $q->whereLike('email', '%john%')->whereLike('username', '%john%');
    })
    ->get();
```

### Query Optimization

Enable automatic query optimization for better performance:

```php
// Enable optimization for a specific query (if available)
$optimizedResults = $db->select('large_table', ['id', 'name'])
    ->where(['status' => 'active'])
    ->optimize()
    ->get();

// Use indexes efficiently with proper WHERE conditions
$results = $db->select('orders', ['*'])
    ->where(['created_at' => ['>=', '2025-01-01']])
    ->orderBy(['created_at' => 'DESC'])
    ->get();

// For MySQL-specific optimizations, use raw SQL when needed
$results = $db->rawQuery(
    "SELECT * FROM orders USE INDEX (idx_created_at) WHERE created_at >= ?",
    ['2025-01-01']
);
```

### Advanced WHERE Conditions

More sophisticated condition building:

```php
// Complex OR grouping
$users = $db->select('users', ['*'])
    ->where(['status' => 'active'])
    ->whereOr(function($query) {
        $query->where(['role' => 'admin']);
        $query->whereLike('permissions', '%manage%');
    })
    ->get();

// Multiple condition operators
$products = $db->select('products', ['*'])
    ->where([
        'price' => ['>', 10],
        'category_id' => ['IN', [1, 2, 3]],
        'status' => 'active'
    ])
    ->whereGreaterThan('stock', 0)
    ->whereLessThan('price', 100)
    ->get();

// JSON column queries (for supported databases like MySQL 5.7+)
$users = $db->select('users', ['*'])
    ->whereRaw('JSON_CONTAINS(preferences, ?)', ['{"notifications": true}'])
    ->whereRaw('JSON_LENGTH(tags) > ?', [0])
    ->get();
```

### Bulk Operations

Efficient handling of large datasets:

```php
// Bulk insert with better performance
$users = [
    ['uuid' => Utils::generateNanoID(), 'name' => 'John', 'email' => 'john@example.com'],
    ['uuid' => Utils::generateNanoID(), 'name' => 'Jane', 'email' => 'jane@example.com'],
    // ... many more records
];

$affectedRows = $db->insertBatch('users', $users);

// Process large datasets in chunks to avoid memory issues
$chunks = array_chunk($users, 1000);
foreach ($chunks as $chunk) {
    $db->insertBatch('users', $chunk);
}

// Bulk update with conditions
$affectedRows = $db->update('users',
    ['status' => 'inactive'],
    ['last_login' => ['<', date('Y-m-d', strtotime('-1 year'))]]
);

// Bulk delete with date conditions
$deleted = $db->delete('logs',
    ['created_at' => ['<', date('Y-m-d', strtotime('-30 days'))]],
    false // Hard delete
);
```

### Query Result Analysis

Analyze and understand query performance:

```php
// Manual performance tracking
$startTime = microtime(true);
$startMemory = memory_get_usage();

$users = $db->select('users', ['*'])
    ->where(['status' => 'active'])
    ->get();

$executionTime = (microtime(true) - $startTime) * 1000;
$memoryUsed = memory_get_usage() - $startMemory;

echo "Execution time: {$executionTime}ms\n";
echo "Memory usage: {$memoryUsed} bytes\n";
echo "Rows returned: " . count($users) . "\n";

// Get execution plan using EXPLAIN (MySQL example)
$plan = $db->rawQuery("EXPLAIN SELECT * FROM users WHERE status = ?", ['active']);

// For complex queries, use EXPLAIN ANALYZE (PostgreSQL) or similar
$complexPlan = $db->rawQuery("
    EXPLAIN SELECT o.*, u.name 
    FROM orders o 
    JOIN users u ON o.user_id = u.id 
    WHERE o.status = ?", 
    ['completed']
);
```
