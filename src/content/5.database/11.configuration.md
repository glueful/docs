This document provides a comprehensive reference for all database configuration options available in Glueful.

## Basic Configuration

The main database configuration is located in `config/database.php`:

```php
<?php

return [
    // Default database connection
    'default' => env('DB_CONNECTION', 'mysql'),

    // Database connections
    'connections' => [
        // MySQL Configuration
        'mysql' => [
            'driver' => 'mysql',
            'host' => env('DB_HOST', 'localhost'),
            'port' => env('DB_PORT', 3306),
            'database' => env('DB_DATABASE', 'glueful'),
            'username' => env('DB_USERNAME', 'root'),
            'password' => env('DB_PASSWORD', ''),
            'charset' => env('DB_CHARSET', 'utf8mb4'),
            'collation' => env('DB_COLLATION', 'utf8mb4_unicode_ci'),
            'prefix' => env('DB_PREFIX', ''),
            'strict' => true,
            'engine' => 'InnoDB',
            'options' => [
                PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
                PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
                PDO::ATTR_EMULATE_PREPARES => false,
                PDO::MYSQL_ATTR_USE_BUFFERED_QUERY => true,
                PDO::MYSQL_ATTR_INIT_COMMAND => 'SET NAMES utf8mb4 COLLATE utf8mb4_unicode_ci',
            ],
        ],

        // PostgreSQL Configuration
        'pgsql' => [
            'driver' => 'pgsql',
            'host' => env('DB_HOST', 'localhost'),
            'port' => env('DB_PORT', 5432),
            'database' => env('DB_DATABASE', 'glueful'),
            'username' => env('DB_USERNAME', 'postgres'),
            'password' => env('DB_PASSWORD', ''),
            'charset' => env('DB_CHARSET', 'utf8'),
            'prefix' => env('DB_PREFIX', ''),
            'schema' => env('DB_SCHEMA', 'public'),
            'sslmode' => env('DB_SSLMODE', 'prefer'),
            'options' => [
                PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
                PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
                PDO::ATTR_EMULATE_PREPARES => false,
            ],
        ],

        // SQLite Configuration
        'sqlite' => [
            'driver' => 'sqlite',
            'database' => env('DB_DATABASE', storage_path('database/database.sqlite')),
            'prefix' => env('DB_PREFIX', ''),
            'foreign_key_constraints' => true,
            'options' => [
                PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
                PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
                PDO::ATTR_EMULATE_PREPARES => false,
            ],
        ],
    ],

    // Migration settings
    'migrations' => [
        'path' => database_path('migrations'),
        'table' => 'migrations',
        'auto_run' => env('DB_AUTO_MIGRATE', false),
        'backup_before_migrate' => env('DB_BACKUP_BEFORE_MIGRATE', true),
    ],
];
```

// Connection pooling configuration
'pooling' => [

        'backoff_multiplier' => env('DB_POOL_BACKOFF_MULTIPLIER', 2),

            'min_connections' => env('DB_POOL_MYSQL_MIN_CONNECTIONS', 5),

        'num_tests_per_eviction_run' => env('DB_POOL_TESTS_PER_EVICTION', 3),
        'remove_abandoned' => env('DB_POOL_REMOVE_ABANDONED', true),
        'remove_abandoned_timeout' => env('DB_POOL_REMOVE_ABANDONED_TIMEOUT', 300), // seconds
        'log_abandoned' => env('DB_POOL_LOG_ABANDONED', true),
    ],
    
    // Pool monitoring and alerts

        'enabled' => env('DB_POOL_MONITORING_ENABLED', true),
        'metrics_collection' => env('DB_POOL_METRICS_COLLECTION', true),
        'detailed_logging' => env('DB_POOL_DETAILED_LOGGING', false),

## Query Caching Configuration

Advanced query result caching for improved performance:

```php
// Query caching configuration
'query_cache' => [
    'enabled' => env('QUERY_CACHE_ENABLED', true),
    'default_ttl' => env('QUERY_CACHE_DEFAULT_TTL', 3600), // seconds
    'max_ttl' => env('QUERY_CACHE_MAX_TTL', 86400), // 24 hours
    'store' => env('QUERY_CACHE_STORE', 'redis'), // redis, memcached, file
    'key_prefix' => env('QUERY_CACHE_KEY_PREFIX', 'query_cache'),
    'auto_invalidate' => env('QUERY_CACHE_AUTO_INVALIDATE', true),
    'invalidate_on_write' => env('QUERY_CACHE_INVALIDATE_ON_WRITE', true),
    
    // Cache exclusions
    'exclude_tables' => [
        'migrations',
        'jobs',
        'failed_jobs',
        'sessions',
        'audit_logs',
    ],
    
    'exclude_patterns' => [
        '/^UPDATE/i',
        '/^INSERT/i',
        '/^DELETE/i',
        '/^ALTER/i',
        '/^DROP/i',
        '/^CREATE/i',
        '/password/i',
        '/token/i',
        '/secret/i',
    ],
    
    // Cache strategies
    'strategies' => [
        'read_through' => env('QUERY_CACHE_READ_THROUGH', true),
        'write_through' => env('QUERY_CACHE_WRITE_THROUGH', false),
        'write_behind' => env('QUERY_CACHE_WRITE_BEHIND', false),
        'refresh_ahead' => env('QUERY_CACHE_REFRESH_AHEAD', false),
    ],
    
    // Cache warming
    'warming' => [
        'enabled' => env('QUERY_CACHE_WARMING_ENABLED', false),
        'queries' => [
            // List of queries to pre-warm
            'SELECT * FROM users WHERE status = "active"',
            'SELECT * FROM settings WHERE active = 1',
        ],
        'schedule' => env('QUERY_CACHE_WARMING_SCHEDULE', '0 6 * * *'), // Daily at 6 AM
    ],
    
    // Cache statistics
    'statistics' => [
        'enabled' => env('QUERY_CACHE_STATISTICS_ENABLED', true),
        'track_hit_ratio' => true,
        'track_memory_usage' => true,
        'track_invalidations' => true,
        'retention_period' => env('QUERY_CACHE_STATS_RETENTION', 2592000), // 30 days
    ],
],
```

## Configuration Validation

Glueful provides configuration validation to ensure your database settings are correct:

```bash
# Validate database configuration
php glueful db:config:validate

# Test database connections
php glueful db:config:test

# Show current configuration
php glueful db:config:show

# Optimize configuration for current environment
php glueful db:config:optimize --environment=production
```
