Glueful provides sophisticated connection pooling capabilities to optimize database performance, manage resources efficiently, and handle high-concurrency scenarios.

## Overview

The connection pooling system offers:

- **Multi-Engine Support**: Different pooling strategies for MySQL, PostgreSQL, SQLite
- **Dynamic Pool Management**: Automatic scaling based on demand
- **Health Monitoring**: Real-time pool health and performance tracking
- **Connection Lifecycle Management**: Automatic cleanup and resource management
- **Failover Support**: Automatic failover to backup connections

## Connection Pool Configuration

## Connection Pool Management

### Basic Pool Operations

```php
use Glueful\Database\ConnectionPoolManager;
use Glueful\Database\Connection;

// Get the pool manager
$poolManager = new ConnectionPoolManager();

// Initialize connection pools
$poolManager->initializePools();

// Get a connection from the pool
$connection = $poolManager->getConnection('mysql');

// Use the connection
$users = $connection->table('users')->get();

// Return connection to pool (automatic with proper resource management)
$poolManager->returnConnection($connection);

// Manual pool operations
$poolManager->expandPool('mysql', 5); // Add 5 more connections
$poolManager->shrinkPool('mysql', 3); // Remove 3 idle connections
$poolManager->resetPool('mysql'); // Reset the entire pool
```

## Advanced Pool Features

### Dynamic Pool Scaling

```php
use Glueful\Database\DynamicPoolScaler;

$scaler = new DynamicPoolScaler($poolManager);

// Configure auto-scaling rules
$scaler->addScalingRule([
    'name' => 'high_utilization_scale_up',
    'trigger' => 'utilization > 80',
    'action' => 'scale_up',
    'amount' => 3, // Add 3 connections
    'cooldown' => 300, // Wait 5 minutes before next scaling
    'max_connections' => 50
]);

$scaler->addScalingRule([
    'name' => 'low_utilization_scale_down',
    'trigger' => 'utilization < 30 AND idle_time > 600',
    'action' => 'scale_down',
    'amount' => 2, // Remove 2 connections
    'cooldown' => 600, // Wait 10 minutes before next scaling
    'min_connections' => 5
]);

// Enable auto-scaling
$scaler->enableAutoScaling();

// Monitor scaling events
$scaler->onScalingEvent(function($event) {
    logger()->info('Pool scaling event', [
        'engine' => $event['engine'],
        'action' => $event['action'],
        'amount' => $event['amount'],
        'reason' => $event['reason']
    ]);
});
```

### Connection Failover

```php
use Glueful\Database\ConnectionFailover;

$failover = new ConnectionFailover();

// Configure failover connections
$failover->addFailoverConnection('mysql', [
    'host' => 'mysql-backup.example.com',
    'database' => 'glueful',
    'username' => 'backup_user',
    'password' => 'backup_password',
    'priority' => 1 // Lower number = higher priority
]);

$failover->addFailoverConnection('mysql', [
    'host' => 'mysql-backup2.example.com',
    'database' => 'glueful',
    'username' => 'backup_user',
    'password' => 'backup_password',
    'priority' => 2
]);

// Configure failover behavior
$failover->configure([
    'max_retry_attempts' => 3,
    'retry_delay' => 1000, // milliseconds
    'health_check_interval' => 30, // seconds
    'automatic_failback' => true,
    'failback_delay' => 300 // seconds
]);

// Enable failover
$poolManager->enableFailover($failover);

// Monitor failover events
$failover->onFailoverEvent(function($event) {
    logger()->error('Database failover occurred', [
        'primary_host' => $event['primary_host'],
        'failover_host' => $event['failover_host'],
        'reason' => $event['reason'],
        'timestamp' => $event['timestamp']
    ]);
});
```

### Connection Load Balancing

```php
use Glueful\Database\ConnectionLoadBalancer;

$loadBalancer = new ConnectionLoadBalancer();

// Configure read replicas
$loadBalancer->addReadReplica('mysql', [
    'host' => 'mysql-read1.example.com',
    'weight' => 3 // Higher weight = more traffic
]);

$loadBalancer->addReadReplica('mysql', [
    'host' => 'mysql-read2.example.com',
    'weight' => 2
]);

// Configure load balancing strategy
$loadBalancer->setStrategy('weighted_round_robin'); // round_robin, least_connections, random

// Separate read and write operations
$writeConnection = $poolManager->getWriteConnection('mysql');
$readConnection = $poolManager->getReadConnection('mysql'); // Automatically load balanced

// Use connections for specific operations
$users = $readConnection->table('users')
    ->where('status', 'active')
    ->get();

$userId = $writeConnection->table('users')
    ->insertGetId(['name' => 'John Doe', 'email' => 'john@example.com']);
```

## Troubleshooting Common Issues

### Connection Pool Exhaustion

```php
// Detect and handle pool exhaustion
$poolManager->onPoolExhaustion(function($engine, $stats) {
    logger()->critical('Connection pool exhausted', [
        'engine' => $engine,
        'active_connections' => $stats['active_connections'],
        'max_connections' => $stats['max_connections'],
        'wait_queue_size' => $stats['wait_queue_size']
    ]);
    
    // Emergency actions
    if ($stats['wait_queue_size'] > 10) {
        // Temporarily increase pool size
        $this->poolManager->emergencyExpansion($engine, 5);
        
        // Alert administrators
        $this->alertManager->sendCriticalAlert([
            'type' => 'pool_exhaustion',
            'engine' => $engine,
            'action_taken' => 'emergency_expansion'
        ]);
    }
});
```

### Connection Leaks

```php
// Detect connection leaks
$leakDetector = new ConnectionLeakDetector($poolManager);

$leakDetector->enableLeakDetection([
    'max_connection_age' => 3600, // 1 hour
    'track_stack_traces' => true,
    'alert_on_leak' => true
]);

// Analyze potential leaks
$leaks = $leakDetector->detectLeaks();

foreach ($leaks as $leak) {
    echo "Potential leak detected:\n";
    echo "Connection ID: {$leak['connection_id']}\n";
    echo "Age: {$leak['age']} seconds\n";
    echo "Created by: {$leak['creator']}\n";
    echo "Stack trace: {$leak['stack_trace']}\n\n";
}

// Automatic leak cleanup
$leakDetector->enableAutoCleanup([
    'max_age' => 1800, // 30 minutes
    'confirm_before_kill' => false
]);
```

### Performance Degradation

```php
// Monitor for performance degradation
$performanceMonitor = new PoolPerformanceMonitor($poolManager);

$performanceMonitor->addThreshold('response_time', [
    'warning' => 500, // ms
    'critical' => 1000 // ms
]);

$performanceMonitor->addThreshold('utilization', [
    'warning' => 70, // %
    'critical' => 90 // %
]);

$performanceMonitor->onThresholdExceeded(function($metric, $level, $value) {
    logger()->warning('Pool performance threshold exceeded', [
        'metric' => $metric,
        'level' => $level,
        'value' => $value
    ]);
    
    if ($level === 'critical') {
        // Take immediate action
        $this->poolManager->optimizePool($metric);
    }
});
```
