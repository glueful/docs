This guide covers best practices and patterns for developing high-quality APIs using the Glueful framework with its modern architecture.

## API Design Principles

Glueful follows RESTful API design principles with modern enhancements leveraging dependency injection, PSR-15 middleware, and enterprise security features.

### RESTful Resource Design

Design your APIs around resources, not actions:

```php
// Good: Resource-oriented URLs
GET    /api/v1/users           // List users
GET    /api/v1/users/123       // Get specific user
POST   /api/v1/users           // Create user
PUT    /api/v1/users/123       // Update user
DELETE /api/v1/users/123       // Delete user

// Avoid: Action-oriented URLs
POST /api/v1/getUser
POST /api/v1/createUser
POST /api/v1/updateUser
```

### HTTP Status Codes

Use appropriate HTTP status codes with Glueful's Response class. Here's an example from the current `AuthController`:

```php
<?php
use Glueful\Http\Response;
use Glueful\Helpers\Request;
use Glueful\Controllers\BaseController;

class AuthController extends BaseController
{
    public function verifyEmail(): Response
    {
        try {
            $postData = Request::getPostData();
            if (!isset($postData['email'])) {
                return Response::error('Email address is required', Response::HTTP_BAD_REQUEST)->send();
            }

            $otp = $this->verifier->generateOTP();
            $result = $this->verifier->sendVerificationEmail($postData['email'], $otp);

            if (!$result['success']) {
                $errorMessage = $result['message'] ?? 'Failed to send verification email';
                $statusCode = isset($result['error_code'])
                    ? Utils::mapErrorCodeToStatusCode($result['error_code'])
                    : Response::HTTP_BAD_REQUEST;

                return Response::error($errorMessage, $statusCode)->send();
            }

            return Response::ok([
                'data' => [
                    'email' => $postData['email'],
                    'expires_in' => EmailVerification::OTP_EXPIRY_MINUTES * 60
                ]
            ], 'Verification code has been sent to your email')->send();
        } catch (\Exception $e) {
            return Response::error(
                'Failed to send verification email: ' . $e->getMessage(),
                Response::HTTP_INTERNAL_SERVER_ERROR
            )->send();
        }
    }

    public function verifyOtp(): Response
    {
        try {
            $postData = Request::getPostData();
            if (!isset($postData['email']) || !isset($postData['otp'])) {
                return Response::error('Email and OTP are required', Response::HTTP_BAD_REQUEST)->send();
            }

            $isValid = $this->verifier->verifyOTP($postData['email'], $postData['otp']);

            if (!$isValid) {
                return Response::error('Invalid or expired OTP', Response::HTTP_BAD_REQUEST)->send();
            }

            return Response::ok([
                'data' => [
                    'email' => $postData['email'],
                    'verified' => true,
                    'verified_at' => date('Y-m-d\TH:i:s\Z')
                ]
            ], 'OTP verified successfully')->send();
        } catch (\Exception $e) {
            return Response::error(
                'Failed to verify OTP: ' . $e->getMessage(),
                Response::HTTP_INTERNAL_SERVER_ERROR
            )->send();
        }
    }
}
```

## Request Validation

Proper input validation is crucial for API security and data integrity. Glueful uses an attribute-based validation system with DTOs.

### DTO-Based Validation (Recommended)

For structured validation, use Data Transfer Objects with attributes:

```php
<?php
use Glueful\Validation\Attributes\{Rules, Sanitize};
use Glueful\Validation\Validator;
use Glueful\Http\Response;

class UserRegistrationDTO
{
    #[Sanitize(['trim', 'strip_tags'])]
    #[Rules(['required', 'string', 'min:3', 'max:50'])]
    public string $name;

    #[Sanitize(['trim', 'sanitize_email'])]
    #[Rules(['required', 'email'])]
    public string $email;

    #[Sanitize(['trim'])]
    #[Rules(['required', 'string', 'min:8'])]
    public string $password;

    #[Sanitize(['intval'])]
    #[Rules(['required', 'int', 'min:18', 'max:99'])]
    public int $age;

    #[Rules(['string', 'in:active,inactive,pending'])]
    public string $status = 'pending';
}

class UserController extends BaseController
{
    public function __construct(
        private UserRepository $userRepository
    ) {
        parent::__construct();
    }

    public function register(): Response
    {
        try {
            $postData = Request::getPostData();

            // Create and populate DTO
            $dto = new UserRegistrationDTO();
            $dto->name = $postData['name'] ?? '';
            $dto->email = $postData['email'] ?? '';
            $dto->password = $postData['password'] ?? '';
            $dto->age = $postData['age'] ?? 0;
            $dto->status = $postData['status'] ?? 'pending';

            // Validate DTO
            $validator = new Validator();
            if (!$validator->validate($dto)) {
                return Response::error(
                    'Validation failed',
                    Response::HTTP_UNPROCESSABLE_ENTITY,
                    Response::ERROR_VALIDATION,
                    null,
                    $validator->errors()
                )->send();
            }

            // DTO now contains sanitized and validated data
            $user = $this->userRepository->create($dto);
            return Response::created($user, 'User registered successfully')->send();

        } catch (\Exception $e) {
            return Response::error(
                'Registration failed: ' . $e->getMessage(),
                Response::HTTP_INTERNAL_SERVER_ERROR
            )->send();
        }
    }
}
```

### Role-Based Authorization

Using the actual `RoleRepository` implementation:

```php
<?php
use Glueful\Repositories\RoleRepository;
use Glueful\Http\Response;
use Glueful\Helpers\Utils;

class AdminController extends BaseController
{
    public function __construct(
        private RoleRepository $roleRepository
    ) {
        parent::__construct();
    }

    public function dashboard(): Response
    {
        // Use BaseController's built-in admin checking
        $this->requireCachedAdmin();
        
        return Response::ok($this->getDashboardData())->send();
    }

    public function createRole(): Response
    {
        // Check admin permission
        $this->requirePermission('roles.create', 'system');
        
        try {
            $postData = Request::getPostData();

            if (empty($postData['name'])) {
                return Response::error(
                    'Role name is required',
                    Response::HTTP_BAD_REQUEST
                )->send();
            }

            // Audit the action
            $roleId = $this->auditResourceAction('roles', 'create', function() use ($postData) {
                return $this->roleRepository->create([
                    'name' => $postData['name'],
                    'description' => $postData['description'] ?? '',
                    'created_at' => date('Y-m-d H:i:s')
                ]);
            });

            return Response::created([
                'role_id' => $roleId,
                'name' => $postData['name']
            ], 'Role created successfully')->send();

        } catch (\Exception $e) {
            return Response::error(
                'Failed to create role: ' . $e->getMessage(),
                Response::HTTP_INTERNAL_SERVER_ERROR
            )->send();
        }
    }
}
```

```php
                'Failed to delete user: ' . $e->getMessage(),
```

## Error Handling

Implement consistent error handling across your API using Glueful's standardized response system.

### Using the Response Class

```php
<?php
use Glueful\Http\Response;

class UserController extends BaseController
{
    public function __construct(
        private UserRepository $userRepository
    ) {
        parent::__construct();
    }

    public function show(string $id): Response
    {
        // Check permission to view users
        $this->requirePermission('users.read', 'users');
        
        try {
            $user = $this->userRepository->findById($id);

            if (!$user) {
                return Response::notFound('User not found')->send();
            }

            return Response::ok($user)->send();

        } catch (\Exception $e) {
            return Response::error(
                'Failed to retrieve user: ' . $e->getMessage(),
                Response::HTTP_INTERNAL_SERVER_ERROR,
                Response::ERROR_SERVER
            )->send();
        }
    }

    public function store(): Response
    {
        // Check permission to create users
        $this->requirePermission('users.create', 'users');
        
        try {
            $postData = Request::getPostData();

            // Validation
            if (empty($postData['email'])) {
                return Response::error(
                    'Email is required',
                    Response::HTTP_BAD_REQUEST,
                    Response::ERROR_VALIDATION
                )->send();
            }

            $user = $this->auditResourceAction('users', 'create', function() use ($postData) {
                return $this->userRepository->create($postData);
            });
            
            return Response::created($user, 'User created successfully')->send();

        } catch (\Exception $e) {
            return Response::error(
                'Failed to create user: ' . $e->getMessage(),
                Response::HTTP_INTERNAL_SERVER_ERROR,
                Response::ERROR_SERVER
            )->send();
        }
    }
}
```

### Global Error Handler

The framework includes built-in error handling in the `ExceptionHandler`:

```php
<?php
// From Glueful\Exceptions\ExceptionHandler
class ExceptionHandler
{
    public static function sendJsonResponse(
        string $message,
        int $statusCode = 500,
        ?array $data = null
    ): void {
        $response = [
            'success' => false,
            'status' => $statusCode,
            'message' => $message
        ];

        // Add data only if it's not null
        if ($data !== null) {
            $response['data'] = $data;
        }

        // Set HTTP response code
        http_response_code($statusCode);

        // Set JSON content type
        header('Content-Type: application/json');

        // Output JSON
        echo json_encode($response, JSON_UNESCAPED_SLASHES | JSON_UNESCAPED_UNICODE);

        exit;
    }
}
```

### Structured Error Responses

The `Response` class provides standardized error types:

```php
<?php
// Available error types in Response class
class Response
{
    public const ERROR_VALIDATION = 'VALIDATION_ERROR';
    public const ERROR_AUTHENTICATION = 'AUTHENTICATION_ERROR';
    public const ERROR_AUTHORIZATION = 'AUTHORIZATION_ERROR';
    public const ERROR_NOT_FOUND = 'NOT_FOUND_ERROR';
    public const ERROR_RATE_LIMIT = 'RATE_LIMIT_ERROR';
    public const ERROR_SERVER = 'SERVER_ERROR';
    public const ERROR_SECURITY = 'SECURITY_ERROR';

    public static function error(
        string $message,
        int $statusCode = self::HTTP_BAD_REQUEST,
        ?string $errorType = null,
        ?string $errorCode = null,
        mixed $details = null,
        ?string $requestId = null
    ): self {
        $errorDetails = [
            'type' => $errorType ?? self::getErrorTypeFromStatus($statusCode),
            'code' => $errorCode,
            'details' => $details,
            'timestamp' => date('c'),
            'request_id' => $requestId ?? self::generateRequestId()
        ];

        // Remove null values from error details
        $errorDetails = array_filter($errorDetails, fn($value) => $value !== null);

        return new self(null, $statusCode, $message, false, $errorDetails);
    }
}
```

```php
<?php
// In your bootstrap file or service provider
use Glueful\Exceptions\ExceptionHandler;

ExceptionHandler::register(function(\Throwable $e) {
    // Log the exception
    error_log("API Error: " . $e->getMessage());

    // Return appropriate response based on exception type
    if ($e instanceof ValidationException) {
        return Response::error(
            $e->getMessage(),
            422,
            'VALIDATION_ERROR',
            null,
            $e->getErrors()
        )->send();
    }

    if ($e instanceof ResourceNotFoundException) {
        return Response::notFound($e->getMessage())->send();
    }

    // Default error response
    return Response::error(
        'Internal server error',
        500
    )->send();
});
```

### Database Query Optimization

```php
class UserRepository extends BaseRepository
{
    protected string $table = 'users';
    
    public function getAll(array $filters = [], int $limit = 20, int $offset = 0): array
    {
        $queryBuilder = $this->getNewQueryBuilder();
        
        // Select only needed columns to reduce memory usage
        $columns = ['uuid', 'username', 'email', 'status', 'created_at'];
        
        $query = $queryBuilder->select($this->table, $columns);

        // Apply filters efficiently
        $conditions = [];
        
        if (isset($filters['status'])) {
            $conditions['status'] = $filters['status'];
        }
        
        // Add conditions if any
        if (!empty($conditions)) {
            $query->where($conditions);
        }

        // Handle search with multiple fields
        if (isset($filters['search'])) {
            $searchTerm = $filters['search'];
            $query->whereRaw(
                "(username LIKE ? OR email LIKE ?)",
                ["%{$searchTerm}%", "%{$searchTerm}%"]
            );
        }

        return $query
            ->orderBy(['created_at' => 'DESC'])
            ->limit($limit)
            ->offset($offset)
            ->get();
    }
    
    public function findActiveUsersWithProfiles(): array
    {
        $queryBuilder = $this->getNewQueryBuilder();
        
        return $queryBuilder
            ->select('users', ['users.uuid', 'users.username', 'users.email', 'profiles.bio'])
            ->join('profiles', 'users.uuid = profiles.user_uuid', 'LEFT')
            ->where(['users.status' => 'active'])
            ->whereNotNull('users.deleted_at') // Exclude soft-deleted
            ->orderBy(['users.created_at' => 'DESC'])
            ->get();
    }
    
    public function searchUsers(string $term, array $fields = ['username', 'email']): array
    {
        $queryBuilder = $this->getNewQueryBuilder();
        
        // Use the built-in search functionality
        return $queryBuilder
            ->select($this->table, ['uuid', 'username', 'email'])
            ->search($fields, $term, 'OR')
            ->where(['status' => 'active'])
            ->limit(50)
            ->get();
    }
    
    public function getUsersWithFilters(array $filters): array
    {
        $queryBuilder = $this->getNewQueryBuilder();
        $query = $queryBuilder->select($this->table, ['*']);
        
        // Advanced filtering with different operators
        if (isset($filters['created_after'])) {
            $query->whereRaw('created_at > ?', [$filters['created_after']]);
        }
        
        if (isset($filters['roles'])) {
            $query->whereIn('role', $filters['roles']);
        }
        
        if (isset($filters['age_range'])) {
            $query->whereBetween('age', $filters['age_range'][0], $filters['age_range'][1]);
        }
        
        return $query->get();
    }
}
```

```php
    {
        // Check permissions
        $this->requirePermission('users.create', 'users');
        
        $data = Request::getPostData();

        // Sanitize input data
        $data = $this->sanitizeInput($data);

        // Validate after sanitization
        $validator = new Validator();
        // ... validation logic
    }

    private function sanitizeInput(array $data): array
    {

            }
            return $value;
        }, $data);
    }
```

### CORS Configuration

Glueful provides a comprehensive CORS handler that supports multiple configuration options for different development scenarios.

#### Basic CORS Setup

```php
use Glueful\Http\Cors;

// Option 1: Use config-based CORS (recommended for production)
$cors = new Cors(); // Uses security config automatically
if (!$cors->handle()) {
    return; // OPTIONS request handled, don't continue to routing
}
```

#### Configuration Options

Configure CORS through your application's security config:

```php
// config/security.php
return [
    'cors' => [
        'allowed_origins' => ['https://yourdomain.com', 'https://app.yourdomain.com'],
        'allowed_methods' => ['GET', 'POST', 'PUT', 'DELETE', 'OPTIONS'],
        'allowed_headers' => ['Content-Type', 'Authorization', 'X-API-Version'],
        'expose_headers' => ['X-API-Version', 'X-Rate-Limit-Remaining'],
        'max_age' => 86400,
        'supports_credentials' => true
    ]
];
```

#### Factory Methods for Common Scenarios

```php
use Glueful\Http\Cors;

// Development setup (allows common localhost ports)
$cors = Cors::development();
$cors->handle();

// Production setup with specific origins
$cors = Cors::production(['https://myapp.com', 'https://api.myapp.com']);
$cors->handle();

// Universal CORS (allows ALL origins - use with caution)
$cors = Cors::universal();
$cors->handle();

// API-only setup
$cors = Cors::apiOnly(['https://trusted-client.com']);
$cors->handle();

// Frontend-specific setup
$cors = Cors::frontend(['https://app.mysite.com', 'https://admin.mysite.com']);
$cors->handle();
```

#### Fluent Interface Configuration

```php
use Glueful\Http\Cors;

// Chain configuration methods
$cors = Cors::development()
    ->allowOrigins(['https://custom-domain.com'])
    ->withCredentials(true)
    ->withDebug(true);

$cors->handle();
```

### API Key Management

```php
<?php
use Glueful\Helpers\Utils;

class ApiKeyController extends BaseController
{
    public function __construct(
        private UserRepository $userRepository
    ) {
        parent::__construct();
    }

    public function generateKey(): Response
    {
        // Require authentication
        $this->requireCachedAuthentication();
        
        $user = $this->getCurrentUser();

        $apiKey = $this->userRepository->generateApiKey($user['uuid'], [
            'name' => 'Production API Key',
            'scopes' => ['read_users', 'write_users'],
            'expires_at' => now()->addYears(1)
        ]);

        return Response::created([
            'api_key' => $apiKey['key'],
            'expires_at' => $apiKey['expires_at']
        ])->send();
    }
}
```

### Performance Monitoring

```php
class PerformanceMiddleware implements MiddlewareInterface
{
    public function process(Request $request, RequestHandlerInterface $handler): Response
    {
        $startTime = microtime(true);
        $startMemory = memory_get_usage();

        $response = $handler->handle($request);

        $executionTime = microtime(true) - $startTime;
        $memoryUsage = memory_get_usage() - $startMemory;

        // Log performance metrics
        Log::info('API Performance', [
            'route' => $request->getPathInfo(),
            'method' => $request->getMethod(),
            'execution_time' => $executionTime,
            'memory_usage' => $memoryUsage,
            'status_code' => $response->getStatusCode()
        ]);

        return $response;
    }
}
```
